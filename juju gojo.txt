local function CreateRedHighlight()
    if redHighlight then redHighlight:Destroy() end
    redHighlight = Instance.new("Highlight")
    redHighlight.FillColor = Color3.fromRGB(255, 0, 0)
    redHighlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    redHighlight.FillTransparency = 0.1
    redHighlight.OutlineTransparency = 0.2
    redHighlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    redHighlight.Enabled = false
    redHighlight.Parent = game:GetService("CoreGui")
end

local function CreateLineDrawing()
    if lineDrawing then lineDrawing:Remove() end
    lineDrawing = Drawing.new("Line")
    lineDrawing.Color = Color3.fromRGB(255, 0, 0)
    lineDrawing.Thickness = 2
    lineDrawing.Transparency = 0.8
    lineDrawing.Visible = false
end

local function RemoveVisuals()
    if redHighlight then
        redHighlight.Enabled = false
        redHighlight.Adornee = nil
    end
    if lineDrawing then
        lineDrawing.Visible = false
    end
end

local function UpdateVisuals(targetPlayer)
    if not redHighlight then CreateRedHighlight() end
    if targetPlayer and targetPlayer.Character then
        redHighlight.Adornee = targetPlayer.Character
        redHighlight.Enabled = true
    else
        redHighlight.Enabled = false
        redHighlight.Adornee = nil
    end
end

local function UpdateLineDrawing(aimPos)
    if not lineDrawing then CreateLineDrawing() end
    if not aimPos or not (SilentAimEnabled or CamLockEnabled) then
        lineDrawing.Visible = false
        return
    end
    local cam = Camera
    if not cam then
        lineDrawing.Visible = false
        return
    end
    local targetScreenPos, targetOnScreen = cam:WorldToScreenPoint(aimPos)
    if targetOnScreen then
        local screenSize = cam.ViewportSize
        local startPos = Vector2.new(screenSize.X / 2, 0)
        lineDrawing.From = startPos
        lineDrawing.To = Vector2.new(targetScreenPos.X, targetScreenPos.Y)
        lineDrawing.Visible = true
    else
        lineDrawing.Visible = false
    end
end

    end
end

local function GetPlayerNames()
    local names = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then table.insert(names, p.Name) end
    end
    table.sort(names)
    return names
end

local function GetPlayerByName(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name == name and p ~= LocalPlayer then return p end
    end
    return nil
end

local function GetClosestPlayerByDistance()
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local root = char.HumanoidRootPart
    local nearest, minDist = nil, LOCK_DISTANCE
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
        if hrp and hum and hum.Health > 0 then
            local d = (hrp.Position - root.Position).Magnitude
            if d < minDist then
                minDist = d
                nearest = p
            end
        end
    end
    return nearest
end

local function CamIsValid(plr)
    if not plr or not plr.Character then return false end
    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
    local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return false end
    if hum.Health <= 0 then return false end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return false end
    return (hrp.Position - myChar.HumanoidRootPart.Position).Magnitude <= CAM_LOCK_DISTANCE
end

local function CamFindTarget()
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local root = myChar.HumanoidRootPart
    local nearest, bestDist = nil, CAM_LOCK_DISTANCE
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local hum = plr.Character:FindFirstChildOfClass("Humanoid")
            local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 then
                local d = (hrp.Position - root.Position).Magnitude
                if d < bestDist then
                    bestDist = d
                    nearest = plr
                end
            end
        end
    end
    return nearest
end

local aimCamConnection = nil

local function aimCamUpdate()
    if SilentAimEnabled then
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            if not CurrentTarget
                or not CurrentTarget.Character
                or not CurrentTarget.Character:FindFirstChild("Humanoid")
                or CurrentTarget.Character.Humanoid.Health <= 0
            then
                CurrentTarget = GetClosestPlayerByDistance()
            end

            if CurrentTarget and CurrentTarget.Character then
                local targetHRP = CurrentTarget.Character:FindFirstChild("HumanoidRootPart")
                if targetHRP then
                    char.HumanoidRootPart.CFrame = CFrame.new(
                        char.HumanoidRootPart.Position,
                        Vector3.new(targetHRP.Position.X, char.HumanoidRootPart.Position.Y, targetHRP.Position.Z)
                    )
                    UpdateVisuals(CurrentTarget)
                    local aimPos = targetHRP.Position + targetHRP.AssemblyLinearVelocity * CAM_PREDICTION
                    UpdateLineDrawing(aimPos)
                end
            else
                RemoveVisuals()
                CurrentTarget = nil
            end
        end
    else
        CurrentTarget = nil
    end

    if CamLockEnabled then
        if not CamIsValid(CamlockTarget) then
            if tick() - CamLastValid < CamGrace then return end
            CamlockTarget = CamFindTarget()
            if not CamlockTarget then
                if not SilentAimEnabled then RemoveVisuals() end
                return
            end
        end

        CamLastValid = tick()
        local hrp = CamlockTarget.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local predicted = hrp.Position + hrp.AssemblyLinearVelocity * CAM_PREDICTION
            local currentCF = Camera.CFrame
            local targetCF = CFrame.new(currentCF.Position, predicted)
            Camera.CFrame = currentCF:Lerp(targetCF, CAM_SMOOTHNESS)

            UpdateVisuals(CamlockTarget)
            UpdateLineDrawing(predicted)
        end
    else
        CamlockTarget = nil
    end

    if not SilentAimEnabled and not CamLockEnabled then
        RemoveVisuals()
    end
end

local function safeSetCamToggle(value)

end

local function updateAimCamConnection()
    if SilentAimEnabled or CamLockEnabled then
        if not aimCamConnection then
            aimCamConnection = RunService.RenderStepped:Connect(aimCamUpdate)
        end
    else
        if aimCamConnection then
            aimCamConnection:Disconnect()
            aimCamConnection = nil
        end
        RemoveVisuals()
    end
end

local function SetSilentAim(value)
    SilentAimEnabled = value

end

local function SetCamLock(value)
    CamLockEnabled = value
    updateAimCamConnection()
end

function OrbitController:Start(plr)
    if not plr or plr == LocalPlayer then return false end
    if self.Enabled then self:Stop() end

    local localHRP = GetHRP(LocalPlayer)
    if not localHRP then
        local character = LocalPlayer.Character
        if character then
            character:WaitForChild("Humanoid", 5)
            localHRP = character:WaitForChild("HumanoidRootPart", 5)
        end
    end

    local targetHRP = GetHRP(plr)
    if not localHRP or not targetHRP then
        warn("[Orbit] Không tìm thấy HRP")
        return false
    end

    self.Enabled = true
    self.Target = plr

    local offsetVec = localHRP.Position - targetHRP.Position
    if offsetVec.Magnitude > 0.01 then
        self.Angle = math.atan2(offsetVec.Z, offsetVec.X)
    else
        self.Angle = 0
    end

    if self.CameraPart then self.CameraPart:Destroy() end
    self.CameraPart = Instance.new("Part")
    self.CameraPart.Name = "OrbitCameraFocus"
    self.CameraPart.Size = Vector3.new(0.1, 0.1, 0.1)
    self.CameraPart.Transparency = 1
    self.CameraPart.CanCollide = false
    self.CameraPart.Anchored = true
    self.CameraPart.CFrame = CFrame.new(targetHRP.Position)
    self.CameraPart.Parent = Workspace

    local cam = Workspace.CurrentCamera
    if cam then
        cam.CameraType = Enum.CameraType.Custom
        cam.CameraSubject = self.CameraPart
    end

    local targetPos = targetHRP.Position
    local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
    local startPos = targetPos + offset
    localHRP.CFrame = CFrame.lookAt(startPos, targetPos)
    localHRP.AssemblyLinearVelocity = Vector3.zero
    localHRP.AssemblyAngularVelocity = Vector3.zero

    self:BeginOrbitLoop(plr)
    return true
end

function OrbitController:BeginOrbitLoop(plr)
    if not self.Enabled then return end

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.PhysicsConnection then self.PhysicsConnection:Disconnect(); self.PhysicsConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end

    self.CameraConnection = RunService.RenderStepped:Connect(function(dt)
        if not self.Enabled then return end
        local target = self.Target
        if not target then return end
        local targetRoot = GetHRP(target)
        if not targetRoot then return end

        self.Angle = (self.Angle + AngleSpeed * dt) % (math.pi * 2)

        local targetPosition = targetRoot.Position
        local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
        local desiredPosition = targetPosition + offset

        OrbitState.TargetPosition = desiredPosition
        OrbitState.TargetLookAt = targetPosition

        if self.CameraPart then
            self.CameraPart.CFrame = CFrame.new(targetPosition)
        end
    end)

    self.PhysicsConnection = RunService.Heartbeat:Connect(function()
        if not self.Enabled then return end
        local myRoot = GetHRP(LocalPlayer)
        local targetRoot = GetHRP(self.Target)
        if not myRoot or not targetRoot then return end

        local desiredPosition = targetRoot.Position + Vector3.new(
            math.cos(self.Angle) * Radius,
            Height,
            math.sin(self.Angle) * Radius
        )
        local lookAt = targetRoot.Position

        myRoot.CFrame = CFrame.lookAt(desiredPosition, lookAt)

        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero

        if RemoteEvent then
            RemoteEvent:FireServer(nil, myRoot.CFrame)
        end
    end)

    if not self.Line then
        self.Line = Drawing.new("Line")
        self.Line.Thickness = 2
        self.Line.Color = Color3.fromRGB(255, 0, 0)
        self.Line.Transparency = 1
        self.Line.Visible = false
    end

    self.LineConnection = RunService.RenderStepped:Connect(function()
        if not self.Enabled or not self.Target then
            self.Line.Visible = false
            return
        end
        local targetRoot = GetHRP(self.Target)
        local myRoot = GetHRP(LocalPlayer)
        local cam = Workspace.CurrentCamera
        if not targetRoot or not myRoot or not cam then
            self.Line.Visible = false
            return
        end
        local myPos, on1 = cam:WorldToViewportPoint(myRoot.Position)
        local tgtPos, on2 = cam:WorldToViewportPoint(targetRoot.Position)
        if on1 and on2 then
            self.Line.Visible = true
            self.Line.From = Vector2.new(myPos.X, myPos.Y)
            self.Line.To = Vector2.new(tgtPos.X, tgtPos.Y)
        else
            self.Line.Visible = false
        end
    end)
end

function OrbitController:Stop()
    self.Enabled = false
    self.Target = nil

    if self.PhysicsConnection then
        self.PhysicsConnection:Disconnect()
        self.PhysicsConnection = nil
    end
    OrbitState.TargetPosition = nil
    OrbitState.TargetLookAt = nil

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end
    if self.Line then self.Line.Visible = false end
    if self.CameraPart then self.CameraPart:Destroy(); self.CameraPart = nil end

    local cam = Workspace.CurrentCamera
    if cam then
        local character = LocalPlayer.Character
        if character then
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            cam.CameraType = Enum.CameraType.Custom
            if humanoid then cam.CameraSubject = humanoid end
        else
            cam.CameraSubject = nil
        end
    end
    LocalPlayer.DevCameraOcclusionMode = defaultOcclusion
end

function OrbitController:DisableMouseSelection()
    if self.MouseMoveConnection then self.MouseMoveConnection:Disconnect(); self.MouseMoveConnection = nil end
    if self.MouseClickConnection then self.MouseClickConnection:Disconnect(); self.MouseClickConnection = nil end
    if self.MouseHighlight then self.MouseHighlight:Destroy(); self.MouseHighlight = nil end
    Mouse.Icon = ""
end

local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera
local RunService = game:GetService("RunService")

local ESPEnabled = false
local MAX_DISTANCE = 600
local EspData = {}

local CharactersFolder = workspace:FindFirstChild("Characters")
if not CharactersFolder then
    warn("Không tìm thấy workspace.Characters! ESP sẽ không hoạt động.")
end

local function GetRealCharacter(player)
    if not CharactersFolder then return nil end
    local char = CharactersFolder:FindFirstChild(player.Name)
    if not char then return nil end
    if not char:FindFirstChild("HumanoidRootPart") then return nil end
    if not char:FindFirstChild("Head") then return nil end
    if not char:FindFirstChildOfClass("Humanoid") then return nil end
    return char
end

local function ClearPlayerESP(player)
    local data = EspData[player]
    if data then
        for _, obj in pairs(data) do
            if obj and obj.Remove then obj:Remove() end
        end
        EspData[player] = nil
    end
end

local function ClearAllESP()
    for player, _ in pairs(EspData) do
        ClearPlayerESP(player)
    end
end

local function CreatePlayerESP(player)
    local data = {}

    local box = Drawing.new("Square")
    box.Filled = false
    box.Color = Color3.fromRGB(255, 0, 0)
    box.Thickness = 2
    data.box = box

    local outline = Drawing.new("Square")
    outline.Filled = false
    outline.Color = Color3.fromRGB(0, 0, 0)
    outline.Thickness = 2
    data.outline = outline

    local bgBar = Drawing.new("Square")
    bgBar.Filled = true
    bgBar.Color = Color3.fromRGB(40, 40, 40)
    data.bgBar = bgBar

    local healthBar = Drawing.new("Square")
    healthBar.Filled = true
    data.healthBar = healthBar

    local nameText = Drawing.new("Text")
    nameText.Size = 14
    nameText.Color = Color3.fromRGB(255, 255, 255)
    nameText.Center = true
    nameText.Outline = true
    nameText.OutlineColor = Color3.fromRGB(0, 0, 0)
    data.nameText = nameText

    local distText = Drawing.new("Text")
    distText.Size = 12
    distText.Color = Color3.fromRGB(255, 255, 0)
    distText.Center = true
    distText.Outline = true
    distText.OutlineColor = Color3.fromRGB(0, 0, 0)
    data.distText = distText

    local snapLine = Drawing.new("Line")
    snapLine.Color = Color3.fromRGB(255, 0, 0)
    snapLine.Thickness = 1
    data.snapLine = snapLine

    EspData[player] = data
    return data
end

local function UpdateESP()
    if not ESPEnabled then
        for _, data in pairs(EspData) do
            for _, obj in pairs(data) do
                obj.Visible = false
            end
        end
        return
    end

    local localRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not localRoot then return end

    local screenSize = Camera.ViewportSize
    local snapOrigin = Vector2.new(screenSize.X / 2, screenSize.Y - 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then
            if EspData[player] then
                for _, obj in pairs(EspData[player]) do
                    obj.Visible = false
                end
            end
            continue
        end

        local char = GetRealCharacter(player)
        if not char then
            ClearPlayerESP(player)
            continue
        end

        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        local head = char:FindFirstChild("Head")
        if not humanoid or not rootPart or not head then
            ClearPlayerESP(player)
            continue
        end

        local data = EspData[player]
        if not data then
            data = CreatePlayerESP(player)
        end

        if humanoid.Health <= 0 then
            for _, obj in pairs(data) do
                obj.Visible = false
            end
            continue
        end

        local distance = (rootPart.Position - localRoot.Position).Magnitude
        if distance > MAX_DISTANCE then
            for _, obj in pairs(data) do
                obj.Visible = false
            end
            continue
        end

        local headPos, headVisible = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
        local hipHeight = humanoid.HipHeight or 2
        local feetPos, feetVisible = Camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, hipHeight + 2, 0))

        local isOnScreen = headVisible and feetVisible and headPos.Z > 0 and feetPos.Z > 0

        if not isOnScreen then
            data.box.Visible = false
            data.outline.Visible = false
            data.bgBar.Visible = false
            data.healthBar.Visible = false
            data.nameText.Visible = false
            data.distText.Visible = false
            data.snapLine.Visible = false
            continue
        end

        local top = math.min(headPos.Y, feetPos.Y)
        local bottom = math.max(headPos.Y, feetPos.Y)
        local height = bottom - top
        local width = math.max(height * 0.6, 24)
        local left = headPos.X - width / 2

        top = math.clamp(top, -10, screenSize.Y + 10)
        bottom = math.clamp(bottom, -10, screenSize.Y + 10)
        left = math.clamp(left, -10, screenSize.X + 10)

        data.box.Visible = true
        data.box.Size = Vector2.new(width, height)
        data.box.Position = Vector2.new(left, top)
        data.box.Thickness = 2

        data.outline.Visible = true
        data.outline.Size = data.box.Size
        data.outline.Position = data.box.Position
        data.outline.Thickness = 2

        local healthPercent = math.max(0, humanoid.Health / humanoid.MaxHealth)
        local barWidth = math.clamp(height / 25, 2, 4)
        local barLeft = left - barWidth - 6

        data.bgBar.Visible = true
        data.bgBar.Size = Vector2.new(barWidth, height)
        data.bgBar.Position = Vector2.new(barLeft, top)

        data.healthBar.Visible = true
        local fillHeight = height * healthPercent
        local fillTop = top + (height - fillHeight)
        data.healthBar.Size = Vector2.new(barWidth, fillHeight)
        data.healthBar.Position = Vector2.new(barLeft, fillTop)
        data.healthBar.Color = Color3.fromRGB(255 * (1 - healthPercent), 255 * healthPercent, 0)

        data.nameText.Visible = true
        data.nameText.Text = player.Name
        data.nameText.Position = Vector2.new(headPos.X, top - 40)

        data.distText.Visible = true
        data.distText.Text = string.format("M:%03d", math.floor(distance))
        data.distText.Position = Vector2.new(headPos.X, top - 22)

        local snapEndX = math.clamp(headPos.X, 0, screenSize.X)
        local snapEndY = math.clamp(bottom, 0, screenSize.Y)

        data.snapLine.Visible = true
        data.snapLine.From = snapOrigin
        data.snapLine.To = Vector2.new(snapEndX, snapEndY)
    end

    for player, _ in pairs(EspData) do
        if not table.find(Players:GetPlayers(), player) then
            ClearPlayerESP(player)
        end
    end
end

SttTab:Section({
    Title = "Fly"
})

local player = game.Players.LocalPlayer
local camera = workspace.CurrentCamera
local runService = game:GetService("RunService")
local uis = game:GetService("UserInputService")

local flying = false
local flySpeed = 50
local flyConnection = nil
local controls = nil

pcall(function()
    local playerModule = require(player:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule"))
    controls = playerModule:GetControls()
end)

local function getMoveVector()
    if controls then
        return controls:GetMoveVector()
    end
    local move = Vector3.zero
    if uis:IsKeyDown(Enum.KeyCode.W) then move += Vector3.new(0, 0, -1) end
    if uis:IsKeyDown(Enum.KeyCode.S) then move += Vector3.new(0, 0, 1) end
    if uis:IsKeyDown(Enum.KeyCode.A) then move += Vector3.new(-1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.D) then move += Vector3.new(1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
    if uis:IsKeyDown(Enum.KeyCode.LeftShift) then move += Vector3.new(0, -1, 0) end
    return move
end

local function startFly()
    if flying then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    flying = true

    hum.AutoRotate = false
    hum.PlatformStand = true

    local bg = Instance.new("BodyGyro")
    bg.P = 9e4
    bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.Parent = root

    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Velocity = Vector3.zero
    bv.Parent = root

    local defaultFOV = camera.FieldOfView
    local maxFOV = 105
    local fovSmooth = 0.1

    flyConnection = runService.Heartbeat:Connect(function()
        if not flying or not char.Parent or not root.Parent or not hum.Parent then
            stopFly()
            return
        end

        local input = getMoveVector()
        local desiredVelocity

        if controls then
            local camRelative = Vector3.new(input.X, input.Y, input.Z)
            desiredVelocity = camera.CFrame:VectorToWorldSpace(camRelative) * flySpeed
        else
            local camLook = camera.CFrame.LookVector
            local camRight = camera.CFrame.RightVector
            local forward = Vector3.new(camLook.X, 0, camLook.Z).Unit
            local right = Vector3.new(camRight.X, 0, camRight.Z).Unit

            desiredVelocity = (right * input.X + forward * (-input.Z) + Vector3.new(0, input.Y, 0)) * flySpeed
        end

        bv.Velocity = bv.Velocity:Lerp(desiredVelocity, 0.25)

        local lookXZ = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z)
        if lookXZ.Magnitude > 0.001 then
            local targetCF = CFrame.lookAt(root.Position, root.Position + lookXZ.Unit)
            bg.CFrame = bg.CFrame:Lerp(targetCF, 0.2)
        end

        local speedMag = bv.Velocity.Magnitude
        local alpha = math.clamp(speedMag / (flySpeed * 1.2), 0, 1)
        local targetFOV = defaultFOV + (maxFOV - defaultFOV) * alpha
        camera.FieldOfView = camera.FieldOfView + (targetFOV - camera.FieldOfView) * fovSmooth
    end)
end

local function stopFly()
    if not flying then return end
    flying = false
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    local char = player.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            for _, obj in ipairs(root:GetChildren()) do
                if obj:IsA("BodyGyro") or obj:IsA("BodyVelocity") then
                    obj:Destroy()
                end
            end
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.AutoRotate = true
        end
    end
    if camera then
        camera.FieldOfView = 70
    end
end

            startFly()
        else
            stopFly()
        end
    end
