            local p = head.Position:Lerp(hrp.Position, 0.4)
            return Vector3.new(p.X, p.Y + CAM_Y_OFFSET, p.Z)
        end
        return (head and head.Position) or (hrp and hrp.Position)
    end

    local function findTargetByFOV()
        local cam = workspace.CurrentCamera
        if not cam then return nil end
        local center = Vector2.new(
            cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
        local bestD, bestPlr = math.huge, nil
        for _, plr in ipairs(Players:GetPlayers()) do
            local char = getAlive(plr)
            if char then
                local part = char:FindFirstChild("Head")
                    or char:FindFirstChild("HumanoidRootPart")
                if part then
                    local sp, onScreen = cam:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local d = (center - Vector2.new(sp.X, sp.Y)).Magnitude
                        if d < bestD and d <= FOV_PIXELS then
                            bestD, bestPlr = d, plr
                        end
                    end
                end
            end
        end
        return bestPlr
    end

    local function findTargetByMouse()
        local origin = Mouse.Hit and Mouse.Hit.Position
        if not origin then return nil end
        local bestD, bestPlr = math.huge, nil
        for _, plr in ipairs(Players:GetPlayers()) do
            local char = getAlive(plr)
            if char then
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local d = (origin - hrp.Position).Magnitude
                    if d < bestD and d <= CAMLOCK_RADIUS then
                        bestD, bestPlr = d, plr
                    end
                end
            end
        end
        return bestPlr
    end

        if line      then line.Visible      = false end
        if fovCircle then fovCircle.Visible = false end

    local function updateVisuals(target, aimPos)
        ensureHighlight()
        if target and target.Character then
            highlight.Adornee = target.Character
            highlight.Enabled = true
        else
            highlight.Enabled = false
            highlight.Adornee = nil
        end

        ensureLine()
        if line and aimPos and (SilentAimEnabled or CamLockEnabled) then

    local function onUpdate()
        if not SilentAimEnabled and not CamLockEnabled then
            clearVisuals()
            return
        end

        local now = tick()
        local dt  = math.clamp(now - lastFrameTime, 0.001, 0.1)
        lastFrameTime = now

        local cam = workspace.CurrentCamera
        if not cam then return end

        if not isUsable(ActiveTarget) then
            ActiveTarget = resolveTarget()
        end

        local char = getAlive(ActiveTarget)
        if not char then
            local didGrace = false

            if SilentAimEnabled and LastSilentPos
                and (now - LastSilentTime) < GRACE_TIME
            then
                local myHRP = getMyHRP()
                if myHRP then
                    local flat = Vector3.new(
                        LastSilentPos.X - myHRP.Position.X, 0,
                        LastSilentPos.Z - myHRP.Position.Z)
                    if flat.Magnitude > MIN_FLAT_DIST then
                        local cf = CFrame.new(myHRP.Position,
                            myHRP.Position + flat.Unit)
                        myHRP.CFrame = myHRP.CFrame:Lerp(cf, ROT_LERP_ALPHA)
                    end
                end
                didGrace = true
            end

            if CamLockEnabled and LastCamPos
                and (now - LastCamTime) < GRACE_TIME
            then
                local cur = cam.CFrame
                if (LastCamPos - cur.Position).Magnitude > MIN_CAM_DIST then
                    cam.CFrame = cur:Lerp(
                        CFrame.new(cur.Position, LastCamPos), CAM_MIN_ALPHA)
                end
                didGrace = true
            end

            if didGrace then
                updateVisuals(ActiveTarget, LastCamPos or LastSilentPos)
                return
            end

            ActiveTarget    = nil
            LastSilentPos   = nil
            LastCamPos      = nil
            clearVisuals()
            return
        end

        local pred = getPrediction()

        if SilentAimEnabled then
            local part = getSilentAimPart(ActiveTarget)
            if part then
                local aimPos = part.Position
                local myHRP  = getMyHRP()
                if myHRP then
                    local flat = Vector3.new(
                        aimPos.X - myHRP.Position.X, 0,
                        aimPos.Z - myHRP.Position.Z)
                    if flat.Magnitude > MIN_FLAT_DIST then
                        local cf = CFrame.new(myHRP.Position,
                            myHRP.Position + flat.Unit)
                        myHRP.CFrame = myHRP.CFrame:Lerp(cf, ROT_LERP_ALPHA)
                    end
                end
                LastSilentPos, LastSilentTime = aimPos, now
            end
        end

        if CamLockEnabled then
            local basePos = getCamAimPos(char)
            if basePos then
                local hrp    = char:FindFirstChild("HumanoidRootPart")
                local aimPos = hrp
                    and (basePos + hrp.AssemblyLinearVelocity * pred)
                    or basePos

                local cur = cam.CFrame
                local dir = aimPos - cur.Position
                local dist = dir.Magnitude

                if dist > MIN_CAM_DIST then
                    local dot   = math.clamp(cur.LookVector:Dot(dir.Unit), -1, 1)
                    local angle = math.acos(dot)
                    local base  = CAM_MIN_ALPHA + (CAM_MAX_ALPHA - CAM_MIN_ALPHA)
                                  * math.clamp(angle / math.pi, 0, 1)
                    local alpha = math.clamp(base * (dt / 0.016), 0.05, 0.7)
                    cam.CFrame = cur:Lerp(CFrame.new(cur.Position, aimPos), alpha)
                end
                LastCamPos, LastCamTime = aimPos, now
            end
        end

        updateVisuals(ActiveTarget, LastCamPos or LastSilentPos)
    end

    local function updateConnection()

            if connUpdate then connUpdate:Disconnect(); connUpdate = nil end
            ActiveTarget, LastSilentPos, LastCamPos = nil, nil, nil
            clearVisuals()
        end
    end

function OrbitController:Start(plr)
    if not plr or plr == LocalPlayer then return false end
    if self.Enabled then self:Stop() end

    local localHRP = GetHRP(LocalPlayer)
    if not localHRP then
        local character = LocalPlayer.Character
        if character then
            character:WaitForChild("Humanoid", 5)
            localHRP = character:WaitForChild("HumanoidRootPart", 5)
        end
    end

    local targetHRP = GetHRP(plr)
    if not localHRP or not targetHRP then
        warn("[Orbit] Không tìm thấy HRP")
        return false
    end

    self.Enabled = true
    self.Target = plr

    local offsetVec = localHRP.Position - targetHRP.Position
    if offsetVec.Magnitude > 0.01 then
        self.Angle = math.atan2(offsetVec.Z, offsetVec.X)
    else
        self.Angle = 0
    end

    if self.CameraPart then self.CameraPart:Destroy() end
    self.CameraPart = Instance.new("Part")
    self.CameraPart.Name = "OrbitCameraFocus"
    self.CameraPart.Size = Vector3.new(0.1, 0.1, 0.1)
    self.CameraPart.Transparency = 1
    self.CameraPart.CanCollide = false
    self.CameraPart.Anchored = true
    self.CameraPart.CFrame = CFrame.new(targetHRP.Position)
    self.CameraPart.Parent = Workspace

    local cam = Workspace.CurrentCamera
    if cam then
        cam.CameraType = Enum.CameraType.Custom
        cam.CameraSubject = self.CameraPart
    end

    local targetPos = targetHRP.Position
    local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
    local startPos = targetPos + offset
    localHRP.CFrame = CFrame.lookAt(startPos, targetPos)
    localHRP.AssemblyLinearVelocity = Vector3.zero
    localHRP.AssemblyAngularVelocity = Vector3.zero

    self:BeginOrbitLoop(plr)
    return true
end

function OrbitController:BeginOrbitLoop(plr)
    if not self.Enabled then return end

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.PhysicsConnection then self.PhysicsConnection:Disconnect(); self.PhysicsConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end

    self.CameraConnection = RunService.RenderStepped:Connect(function(dt)
        if not self.Enabled then return end
        local target = self.Target
        if not target then return end
        local targetRoot = GetHRP(target)
        if not targetRoot then return end

        self.Angle = (self.Angle + AngleSpeed * dt) % (math.pi * 2)

        local targetPosition = targetRoot.Position
        local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
        local desiredPosition = targetPosition + offset

        OrbitState.TargetPosition = desiredPosition
        OrbitState.TargetLookAt = targetPosition

        if self.CameraPart then
            self.CameraPart.CFrame = CFrame.new(targetPosition)
        end
    end)

    self.PhysicsConnection = RunService.Heartbeat:Connect(function()
        if not self.Enabled then return end
        local myRoot = GetHRP(LocalPlayer)
        local targetRoot = GetHRP(self.Target)
        if not myRoot or not targetRoot then return end

        local desiredPosition = targetRoot.Position + Vector3.new(
            math.cos(self.Angle) * Radius,
            Height,
            math.sin(self.Angle) * Radius
        )
        local lookAt = targetRoot.Position

        myRoot.CFrame = CFrame.lookAt(desiredPosition, lookAt)

        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero

        if RemoteEvent then
            RemoteEvent:FireServer(nil, myRoot.CFrame)
        end
    end)

    if not self.Line then
        self.Line = Drawing.new("Line")
        self.Line.Thickness = 2
        self.Line.Color = Color3.fromRGB(255, 0, 0)
        self.Line.Transparency = 1
        self.Line.Visible = false
    end

    self.LineConnection = RunService.RenderStepped:Connect(function()
        if not self.Enabled or not self.Target then
            self.Line.Visible = false
            return
        end
        local targetRoot = GetHRP(self.Target)
        local myRoot = GetHRP(LocalPlayer)
        local cam = Workspace.CurrentCamera
        if not targetRoot or not myRoot or not cam then
            self.Line.Visible = false
            return
        end
        local myPos, on1 = cam:WorldToViewportPoint(myRoot.Position)
        local tgtPos, on2 = cam:WorldToViewportPoint(targetRoot.Position)
        if on1 and on2 then
            self.Line.Visible = true
            self.Line.From = Vector2.new(myPos.X, myPos.Y)
            self.Line.To = Vector2.new(tgtPos.X, tgtPos.Y)
        else
            self.Line.Visible = false
        end
    end)
end

function OrbitController:Stop()
    self.Enabled = false
    self.Target = nil

    if self.PhysicsConnection then
        self.PhysicsConnection:Disconnect()
        self.PhysicsConnection = nil
    end
    OrbitState.TargetPosition = nil
    OrbitState.TargetLookAt = nil

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end
    if self.Line then self.Line.Visible = false end
    if self.CameraPart then self.CameraPart:Destroy(); self.CameraPart = nil end

    local cam = Workspace.CurrentCamera
    if cam then
        local character = LocalPlayer.Character
        if character then
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            cam.CameraType = Enum.CameraType.Custom
            if humanoid then cam.CameraSubject = humanoid end
        else
            cam.CameraSubject = nil
        end
    end
    LocalPlayer.DevCameraOcclusionMode = defaultOcclusion
end

    State.Character = character

    State.RootPart = character:FindFirstChild("HumanoidRootPart")

local function FireReset()
    local success = pcall(function()
        local Knit = ReplicatedStorage:WaitForChild("Knit", 5)
        local KnitCore = Knit:WaitForChild("Knit", 5)
        local Services = KnitCore:WaitForChild("Services", 5)
        local JoinService = Services:WaitForChild("JoinService", 5)
        local RE = JoinService:WaitForChild("RE", 5)
        local Reset = RE:WaitForChild("Reset", 5)

local function WaitForNewCharacter(oldCharacter)
    local newCharacter
    if Player.Character and Player.Character ~= oldCharacter then
        newCharacter = Player.Character
    else
        local connection
        local finished = false
        connection = Player.CharacterAdded:Connect(function(character)
            if not finished then
                finished = true
                newCharacter = character
            end
        end)
        local start = os.clock()
        while not newCharacter and os.clock() - start < 10 do
            if not State.Enabled then
                break
            end
            task.wait(0.05)
        end
        finished = true
        connection:Disconnect()
    end
    if not newCharacter then
        return nil
    end
    local humanoid = newCharacter:WaitForChild("Humanoid", 10)
    local root = newCharacter:WaitForChild("HumanoidRootPart", 10)
    if not humanoid or not root then
        return nil
    end
    local start = os.clock()
    while humanoid.Health <= 0 and os.clock() - start < 5 do
        if not State.Enabled then
            return nil
        end
        task.wait(0.05)
    end
    if humanoid.Health <= 0 then
        return nil
    end
    UpdateCharacter(newCharacter)
    return newCharacter
end

local function TweenToTarget(character)
    local humanoid = GetHumanoid(character)
    local root = GetRoot(character)
    if not humanoid or not root then
        return false
    end
    if humanoid.Health <= 0 then
        return false
    end
    local target = CFrame.new(CONFIG.TargetPosition)
    local tweenInfo = TweenInfo.new(CONFIG.TweenTime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out, 0, false, 0)
    local success = pcall(function()
        local tween = TweenService:Create(root, tweenInfo, { CFrame = target })
        tween:Play()
        tween.Completed:Wait()
    end)
    return success
end

local function ValidateStable(character)
    local humanoid = GetHumanoid(character)
    local root = GetRoot(character)
    if not humanoid or not root then
        return false
    end
    local startTime = os.clock()
    while os.clock() - startTime < CONFIG.StableTime do
        if not State.Enabled then
            return false
        end
        if Player.Character ~= character then
            return false
        end
        if humanoid.Health <= 0 then
            return false
        end
        if not IsAtTarget(root) then
            return false
        end
        task.wait(CONFIG.CheckInterval)
    end
    if Player.Character ~= character then
        return false
    end
    if humanoid.Health <= 0 then
        return false
    end
    return IsAtTarget(root)
end

local function ExecuteCycle()
    State.Cycle += 1
    local oldCharacter = GetCharacter()
    local oldHumanoid = GetHumanoid(oldCharacter)
    if not oldCharacter or not oldHumanoid or oldHumanoid.Health <= 0 then
        return false
    end
    if not FireReset() then
        return false
    end
    local newCharacter = WaitForNewCharacter(oldCharacter)
    if not newCharacter then
        return false
    end
    local delayStart = os.clock()
    while os.clock() - delayStart < CONFIG.RespawnDelay do
        if not State.Enabled then
            return false
        end
        local humanoid = GetHumanoid(newCharacter)
        if not humanoid or humanoid.Health <= 0 then
            return false
        end
        task.wait(0.05)
    end
    if not TweenToTarget(newCharacter) then
        return false
    end
    local root = GetRoot(newCharacter)
    if not root then
        return false
    end
    if not IsAtTarget(root) then
        return false
    end
    return ValidateStable(newCharacter)
end

local function StartController()
    if State.Running then
        return
    end
    State.Running = true
    State.Cycle = 0
    task.spawn(function()
        while State.Enabled do
            local success = ExecuteCycle()
            if success then
                State.Enabled = false
                Notify("✓ Auto Safe Reset", "Vị trí ổn định trong 3 giây.")
                break
            end
            if State.Enabled then
                task.wait(CONFIG.RetryDelay)
            end
        end
        State.Running = false
    end)
end

-- Ẩn toàn bộ ESP của một player (không xoá)
local function HideESP(data)
    if not data then return end
    if data.container then data.container.Visible = false end
    if data.highlight then data.highlight.Enabled = false end
    if data.snapLine then data.snapLine.Visible = false end
end

-- Xoá ESP khi player rời game
local function ClearPlayerESP(player)
    local data = EspData[player]
    if not data then return end

    if data.container then
        pcall(function() data.container:Destroy() end)
    end
    if data.highlight then
        pcall(function() data.highlight:Destroy() end)
    end
    if data.snapLine then
        pcall(function() data.snapLine:Remove() end)
    end
    EspData[player] = nil
end

local function ClearAllESP()
    for player in pairs(EspData) do
        ClearPlayerESP(player)
    end
end

-- Main update loop
local function UpdateESP()
    if not ESPEnabled then return end

    local folder = RefreshCharactersFolder()
    if not folder then return end

    local localChar = LocalPlayer.Character
    if not localChar then return end
    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    if not localRoot then return end

    local screenSize = Camera.ViewportSize
    local snapOrigin = Vector2.new(screenSize.X / 2, screenSize.Y - 2)

    local seenPlayers = {}

    for _, player in ipairs(Players:GetPlayers()) do
        seenPlayers[player] = true

        -- Bỏ qua chính mình
        if player == LocalPlayer then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        -- Lấy character thật
        local char = GetRealCharacter(player)
        if not char then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        local head = char:FindFirstChild("Head")
        if not humanoid or not rootPart or not head then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        if humanoid.Health <= 0 then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        local distance = (rootPart.Position - localRoot.Position).Magnitude
        if distance > MAX_DISTANCE then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        -- Tạo/lấy data
        local data = EspData[player]
        if not data then
            data = CreatePlayerESP(player)
        end

        -- Hiện container
        data.container.Visible = true

        -- Highlight (chỉ đổi khi character thay đổi)
        if data.character ~= char then
            data.character = char
            data.highlight.Adornee = char
            data.highlight.Enabled = true
        elseif not data.highlight.Enabled then
            data.highlight.Enabled = true
        end

        -- SnapLine (Drawing.Line)
        local rootScreen = Camera:WorldToViewportPoint(rootPart.Position)
        if rootScreen.Z > 0 then
            local targetX = math.clamp(rootScreen.X, 0, screenSize.X)
            local targetY = math.clamp(rootScreen.Y, 0, screenSize.Y)
            data.snapLine.Visible = true
            data.snapLine.From = snapOrigin
            data.snapLine.To = Vector2.new(targetX, targetY)
        else
            data.snapLine.Visible = false
        end

        -- Tính toán Box/Health/Name/Distance
        local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
        local hipHeight = humanoid.HipHeight or 2
        local feetPos = Camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, hipHeight + 2, 0))

        if headPos.Z <= 0 and feetPos.Z <= 0 then
            data.box.Visible = false
            data.outline.Visible = false
            data.bgBar.Visible = false
            data.healthBar.Visible = false
            data.nameText.Visible = false
            data.distText.Visible = false
            continue
        end

        local top = math.min(headPos.Y, feetPos.Y)
        local bottom = math.max(headPos.Y, feetPos.Y)
        local height = math.max(bottom - top, 20)
        local width = math.max(height * 0.6, 24)
        local left = headPos.X - width / 2

        top = math.clamp(top, -50, screenSize.Y + 50)
        left = math.clamp(left, -50, screenSize.X + 50)

        -- Box
        data.box.Visible = true
        data.box.Position = UDim2.fromOffset(left, top)
        data.box.Size = UDim2.fromOffset(width, height)

        -- Outline
        data.outline.Visible = true
        data.outline.Position = UDim2.fromOffset(left - 2, top - 2)
        data.outline.Size = UDim2.fromOffset(width + 4, height + 4)

        -- Health bar
        local maxHealth = math.max(humanoid.MaxHealth, 1)
        local healthPercent = math.clamp(humanoid.Health / maxHealth, 0, 1)
        local barWidth = math.clamp(height / 25, 2, 4)
        local barLeft = left - barWidth - 6

        data.bgBar.Visible = true
        data.bgBar.Position = UDim2.fromOffset(barLeft, top)
        data.bgBar.Size = UDim2.fromOffset(barWidth, height)

        local fillHeight = height * healthPercent
        data.healthBar.Visible = true
        data.healthBar.Position = UDim2.fromOffset(barLeft, top + height - fillHeight)
        data.healthBar.Size = UDim2.fromOffset(barWidth, fillHeight)
        data.healthBar.BackgroundColor3 = Color3.fromRGB(
            255 * (1 - healthPercent),
            255 * healthPercent,
            0
        )

        -- Name
        data.nameText.Visible = true
        data.nameText.Text = player.Name
        data.nameText.Position = UDim2.fromOffset(headPos.X, top - 2)

        -- Distance
        data.distText.Visible = true
        data.distText.Text = string.format("M:%03d", math.floor(distance))
        data.distText.Position = UDim2.fromOffset(headPos.X, top + 16)
    end

    -- Xoá player đã rời
    for player in pairs(EspData) do
        if not seenPlayers[player] then
            ClearPlayerESP(player)
        end
    end
end

local player = game.Players.LocalPlayer
local camera = workspace.CurrentCamera
local runService = game:GetService("RunService")
local uis = game:GetService("UserInputService")

local flying = false
local flySpeed = 50
local flyConnection = nil
local controls = nil

pcall(function()
    local playerModule = require(player:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule"))
    controls = playerModule:GetControls()
end)

local function getMoveVector()
    if controls then
        return controls:GetMoveVector()
    end
    local move = Vector3.zero
    if uis:IsKeyDown(Enum.KeyCode.W) then move += Vector3.new(0, 0, -1) end
    if uis:IsKeyDown(Enum.KeyCode.S) then move += Vector3.new(0, 0, 1) end
    if uis:IsKeyDown(Enum.KeyCode.A) then move += Vector3.new(-1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.D) then move += Vector3.new(1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
    if uis:IsKeyDown(Enum.KeyCode.LeftShift) then move += Vector3.new(0, -1, 0) end
    return move
end

local function startFly()
    if flying then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    flying = true

    hum.AutoRotate = false
    hum.PlatformStand = true

    local bg = Instance.new("BodyGyro")
    bg.P = 9e4
    bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.Parent = root

    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Velocity = Vector3.zero
    bv.Parent = root

    local defaultFOV = camera.FieldOfView
    local maxFOV = 105
    local fovSmooth = 0.1

    flyConnection = runService.Heartbeat:Connect(function()
        if not flying or not char.Parent or not root.Parent or not hum.Parent then
            stopFly()
            return
        end

        local input = getMoveVector()
        local desiredVelocity

        if controls then
            local camRelative = Vector3.new(input.X, input.Y, input.Z)
            desiredVelocity = camera.CFrame:VectorToWorldSpace(camRelative) * flySpeed
        else
            local camLook = camera.CFrame.LookVector
            local camRight = camera.CFrame.RightVector
            local forward = Vector3.new(camLook.X, 0, camLook.Z).Unit
            local right = Vector3.new(camRight.X, 0, camRight.Z).Unit

            desiredVelocity = (right * input.X + forward * (-input.Z) + Vector3.new(0, input.Y, 0)) * flySpeed
        end

        bv.Velocity = bv.Velocity:Lerp(desiredVelocity, 0.25)

        local lookXZ = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z)
        if lookXZ.Magnitude > 0.001 then
            local targetCF = CFrame.lookAt(root.Position, root.Position + lookXZ.Unit)
            bg.CFrame = bg.CFrame:Lerp(targetCF, 0.2)
        end

        local speedMag = bv.Velocity.Magnitude
        local alpha = math.clamp(speedMag / (flySpeed * 1.2), 0, 1)
        local targetFOV = defaultFOV + (maxFOV - defaultFOV) * alpha
        camera.FieldOfView = camera.FieldOfView + (targetFOV - camera.FieldOfView) * fovSmooth
    end)
end

local function stopFly()
    if not flying then return end
    flying = false
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    local char = player.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            for _, obj in ipairs(root:GetChildren()) do
                if obj:IsA("BodyGyro") or obj:IsA("BodyVelocity") then
                    obj:Destroy()
                end
            end
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.AutoRotate = true
        end
    end
    if camera then
        camera.FieldOfView = 70
    end
end

local isHiding = false
local hiddenData = nil
local originalParent = nil

local function hideData()
	if not isHiding then
		return
	end

	local map = workspace:FindFirstChild("Map")
	if not map then
		return
	end
	if hiddenData then
		return
	end

	local data = map:FindFirstChild("Data")
	if not data then
		return
	end

	hiddenData = data
	originalParent = map

	data.Parent = nil
end

local function restoreData()
	if not hiddenData then
		return
	end

	local map = workspace:FindFirstChild("Map")
	if map then
		hiddenData.Parent = map
	elseif originalParent and originalParent.Parent then
		hiddenData.Parent = originalParent
	end

	hiddenData = nil
	originalParent = nil
end

task.spawn(function()
	while task.wait(0.3) do
		if isHiding and not hiddenData then
			hideData()
		end
	end
end)
